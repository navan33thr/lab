
#include <stdio.h>
#include <string.h>

char days[7][12] = {
    "Monday", "Tuesday", "Wednesday",
    "Thursday", "Friday", "Saturday", "Sunday"
};

typedef struct {
    int dd, mm, yy;
} Date;

struct Day {
    char day[12];
    Date date;                  // Nested structure
    char activity[50];
} planner[7];

void read() {
    printf("\n--- Enter Weekly Schedule ---\n");

    for (int i = 0; i < 7; i++) {
        strcpy(planner[i].day, days[i]);

        printf("\n%s - Enter date (dd/mm/yyyy): ",
               planner[i].day);

        scanf("%d/%d/%d",
              &planner[i].date.dd,
              &planner[i].date.mm,
              &planner[i].date.yy);

        getchar();  // Remove newline

        printf("Activity: ");
        fgets(planner[i].activity,
              sizeof(planner[i].activity), stdin);

        planner[i].activity[
            strcspn(planner[i].activity, "\n")
        ] = '\0';
    }

    printf("\nSchedule saved successfully!\n");
}

void display() {
    printf("\n========== WEEKLY REPORT ==========\n");

    printf("%-12s %-12s %s\n",
           "Day", "Date", "Activity");

    printf("-----------------------------------\n");

    for (int i = 0; i < 7; i++) {
        printf("%-12s %02d/%02d/%04d  %s\n",
               planner[i].day,
               planner[i].date.dd,
               planner[i].date.mm,
               planner[i].date.yy,
               planner[i].activity);
    }

    printf("===================================\n");
}

int main() {
    int choice;

    printf("=== WEEKLY PLANNER ===\n");

    while (1) {
        printf("\n1. Enter Schedule");
        printf("\n2. Display Report");
        printf("\n3. Exit");
        printf("\nChoice: ");

        scanf("%d", &choice);

        switch (choice) {
            case 1:
                read();
                break;

            case 2:
                display();
                break;

            case 3:
                printf("Goodbye!\n");
                return 0;

            default:
                printf("Invalid choice!\n");
        }
    }

    return 0;
}
